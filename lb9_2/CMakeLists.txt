cmake_minimum_required(VERSION 3.16)
project(GildedRoseDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(core src/GildedRose.cpp src/QualityRules.cpp)
target_include_directories(core PUBLIC include)
add_executable(GildedRoseDemo src/main.cpp)
target_link_libraries(GildedRoseDemo PRIVATE core)
enable_testing()
add_executable(GildedRoseDemoTests tests/tests.cpp)
target_link_libraries(GildedRoseDemoTests PRIVATE core)
add_test(NAME GildedRoseDemoTests COMMAND GildedRoseDemoTests)
