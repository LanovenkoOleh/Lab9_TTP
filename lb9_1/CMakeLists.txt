cmake_minimum_required(VERSION 3.16)
project(LibraryDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(core src/Library.cpp)
target_include_directories(core PUBLIC include)
add_executable(LibraryDemo src/main.cpp)
target_link_libraries(LibraryDemo PRIVATE core)
enable_testing()
add_executable(LibraryDemoTests tests/tests.cpp)
target_link_libraries(LibraryDemoTests PRIVATE core)
add_test(NAME LibraryDemoTests COMMAND LibraryDemoTests)
